// 캔들 표기법 변환 프로그램 (TypeScript)

interface Candle {
    date: string;
    time: string;
    open: number;
    high: number;
    low: number;
    close: number;
    jdiff_vol: number;
    value: number;
    jongchk: number;
    rate: string;
    sign: string;
}

interface CandleResult {
    date: string;
    time: string;
    candle_code: string;
    relation_code: string;
    candle_char: string;
    relation_char: string;
    hangul: string;
    raw: Candle;
}

// 한글 조합 함수
function combineHangul(chosung: string, jungsung: string, jongsung: string): string {
    const CHOSUNG_LIST = ['ㄱ', 'ㄲ', 'ㄴ', 'ㄷ', 'ㄸ', 'ㄹ', 'ㅁ', 'ㅂ', 'ㅃ', 'ㅅ', 'ㅆ', 'ㅇ', 'ㅈ', 'ㅉ', 'ㅊ', 'ㅋ', 'ㅌ', 'ㅍ', 'ㅎ'];
    const JUNGSUNG_LIST = ['ㅏ', 'ㅐ', 'ㅑ', 'ㅒ', 'ㅓ', 'ㅔ', 'ㅕ', 'ㅖ', 'ㅗ', 'ㅘ', 'ㅙ', 'ㅚ', 'ㅛ', 'ㅜ', 'ㅝ', 'ㅞ', 'ㅟ', 'ㅠ', 'ㅡ', 'ㅢ', 'ㅣ'];
    const JONGSUNG_LIST = ['', 'ㄱ', 'ㄲ', 'ㄳ', 'ㄴ', 'ㄵ', 'ㄶ', 'ㄷ', 'ㄹ', 'ㄺ', 'ㄻ', 'ㄼ', 'ㄽ', 'ㄾ', 'ㄿ', 'ㅀ', 'ㅁ', 'ㅂ', 'ㅄ', 'ㅅ', 'ㅆ', 'ㅇ', 'ㅈ', 'ㅊ', 'ㅋ', 'ㅌ', 'ㅍ', 'ㅎ'];

    const choIdx = CHOSUNG_LIST.indexOf(chosung);
    const jungIdx = JUNGSUNG_LIST.indexOf(jungsung);
    const jongIdx = JONGSUNG_LIST.indexOf(jongsung);

    const unicodeVal = 0xAC00 + (choIdx * 21 * 28) + (jungIdx * 28) + jongIdx;
    return String.fromCharCode(unicodeVal);
}

// 한글 분해 함수
function decomposeHangul(char: string): { chosung: string; jungsung: string; jongsung: string } {
    const CHOSUNG_LIST = ['ㄱ', 'ㄲ', 'ㄴ', 'ㄷ', 'ㄸ', 'ㄹ', 'ㅁ', 'ㅂ', 'ㅃ', 'ㅅ', 'ㅆ', 'ㅇ', 'ㅈ', 'ㅉ', 'ㅊ', 'ㅋ', 'ㅌ', 'ㅍ', 'ㅎ'];
    const JUNGSUNG_LIST = ['ㅏ', 'ㅐ', 'ㅑ', 'ㅒ', 'ㅓ', 'ㅔ', 'ㅕ', 'ㅖ', 'ㅗ', 'ㅘ', 'ㅙ', 'ㅚ', 'ㅛ', 'ㅜ', 'ㅝ', 'ㅞ', 'ㅟ', 'ㅠ', 'ㅡ', 'ㅢ', 'ㅣ'];
    const JONGSUNG_LIST = ['', 'ㄱ', 'ㄲ', 'ㄳ', 'ㄴ', 'ㄵ', 'ㄶ', 'ㄷ', 'ㄹ', 'ㄺ', 'ㄻ', 'ㄼ', 'ㄽ', 'ㄾ', 'ㄿ', 'ㅀ', 'ㅁ', 'ㅂ', 'ㅄ', 'ㅅ', 'ㅆ', 'ㅇ', 'ㅈ', 'ㅊ', 'ㅋ', 'ㅌ', 'ㅍ', 'ㅎ'];

    const code = char.charCodeAt(0) - 0xAC00;
    const choIdx = Math.floor(code / (21 * 28));
    const jungIdx = Math.floor((code % (21 * 28)) / 28);
    const jongIdx = code % 28;

    return {
        chosung: CHOSUNG_LIST[choIdx],
        jungsung: JUNGSUNG_LIST[jungIdx],
        jongsung: JONGSUNG_LIST[jongIdx] || ''
    };
}

// 한글 코드를 텍스트 설명으로 변환하는 함수
export function decodeHangulCandle(hangulCode: string): string {
    // 스페이스 제거하고 처리
    const cleanCode = hangulCode.replace(/\s+/g, '');

    if (cleanCode.length % 2 !== 0) {
        return '오류: 한글 코드는 2글자 단위여야 합니다.';
    }

    const results: string[] = [];
    const errors: string[] = [];
    let candleIndex = 1;

    for (let i = 0; i < cleanCode.length; i += 2) {
        const candleChar = cleanCode[i];
        const relationChar = cleanCode[i + 1];

        // 캔들 자체 디코딩
        const candleDecomp = decomposeHangul(candleChar);
        const V = V_CHOSUNG_REV[candleDecomp.chosung];
        const D = D_JUNGSUNG_REV[candleDecomp.jungsung];
        const S = S_JONGSUNG_REV[candleDecomp.jongsung];

        // 캔들 에러 체크
        const candleErrors: string[] = [];
        if (!V) candleErrors.push(`초성 '${candleDecomp.chosung}'은(는) 유효하지 않은 변동성 코드입니다`);
        if (!D) candleErrors.push(`중성 '${candleDecomp.jungsung}'은(는) 유효하지 않은 방향성 코드입니다`);
        if (!S) candleErrors.push(`종성 '${candleDecomp.jongsung || '(없음)'}' 은(는) 유효하지 않은 꼬리 코드입니다`);

        // 관계 디코딩
        const relationDecomp = decomposeHangul(relationChar);
        const PB = PB_CHOSUNG_REV[relationDecomp.chosung];
        const Q = Q_JUNGSUNG_REV[relationDecomp.jungsung];
        const PR = PR_JONGSUNG_REV[relationDecomp.jongsung];

        // 관계 에러 체크
        const relationErrors: string[] = [];
        if (!PB) relationErrors.push(`초성 '${relationDecomp.chosung}'은(는) 유효하지 않은 몸통위치 코드입니다`);
        if (!Q) relationErrors.push(`중성 '${relationDecomp.jungsung}'은(는) 유효하지 않은 거래량 코드입니다`);
        if (!PR) relationErrors.push(`종성 '${relationDecomp.jongsung || '(없음)'}' 은(는) 유효하지 않은 레인지위치 코드입니다`);

        // 에러가 있으면 에러 메시지 추가
        if (candleErrors.length > 0 || relationErrors.length > 0) {
            errors.push(`\n[캔들${candleIndex}] ${candleChar}${relationChar} - 해석 오류`);
            if (candleErrors.length > 0) {
                errors.push(`  캔들 글자 '${candleChar}' 오류:`);
                candleErrors.forEach(err => errors.push(`    - ${err}`));
            }
            if (relationErrors.length > 0) {
                errors.push(`  관계 글자 '${relationChar}' 오류:`);
                relationErrors.forEach(err => errors.push(`    - ${err}`));
            }
        } else {
            // 에러가 없으면 정상 출력
            const candleDesc = [
                V_DESC[V],
                D_DESC[D],
                S_DESC[S]
            ].join(', ');

            const relationDesc = [
                PB_DESC[PB],
                Q_DESC[Q],
                PR_DESC[PR]
            ].join(', ');

            results.push(`[캔들${candleIndex}] ${candleChar}${relationChar}`);
            results.push(`  캔들: ${candleDesc}`);
            results.push(`  관계: ${relationDesc}`);
            results.push('');
        }

        candleIndex++;
    }

    // 에러가 있으면 에러만 반환
    if (errors.length > 0) {
        return `해석 오류가 발견되었습니다:\n${errors.join('\n')}`;
    }

    return results.join('\n');
}

// 2. 캔들 분석 함수들

function getVCode(candle: Candle, rangeMa: number): string {
    const rangeVal = candle.high - candle.low;
    if (rangeMa === 0) return 'V2';

    const rangeRel = (rangeVal / rangeMa) * 100;

    if (rangeRel < 50) return 'V1';
    if (rangeRel < 150) return 'V2';
    if (rangeRel < 250) return 'V3';
    return 'V4';
}

function getDCode(candle: Candle): string {
    const { open: O, close: C, high: H, low: L } = candle;
    const body = Math.abs(C - O);
    const rangeVal = H - L;

    if (rangeVal === 0) return 'D0';

    const bodyPct = (body / rangeVal) * 100;
    const bodyOverO = O > 0 ? (body / O) * 100 : 0;

    // 도지 판정
    if (bodyPct <= 2 || bodyOverO <= 0.5) return 'D0';

    // 양봉/음봉 구분
    if (C > O) {  // 양봉
        if (bodyPct <= 40) return 'D1';
        if (bodyPct <= 70) return 'D2';
        return 'D3';
    } else {  // 음봉
        if (bodyPct <= 40) return 'D4';
        if (bodyPct <= 70) return 'D5';
        return 'D6';
    }
}

function getSCode(candle: Candle): string {
    const { open: O, close: C, high: H, low: L } = candle;
    const rangeVal = H - L;

    if (rangeVal === 0) return 'S0';

    const U = H - Math.max(O, C);  // 윗꼬리
    const Lw = Math.min(O, C) - L;  // 아랫꼬리

    const uPct = (U / rangeVal) * 100;
    const lwPct = (Lw / rangeVal) * 100;

    // 꼬리 거의 없음
    if (uPct <= 5 && lwPct <= 5) return 'S0';

    // 양쪽 비슷
    const maxShadow = Math.max(U, Lw);
    if (maxShadow > 0) {
        const shadowSimRatio = (Math.abs(U - Lw) / maxShadow) * 100;
        if (shadowSimRatio <= 20) return 'S1';
    }

    // 윗꼬리 우세 vs 아랫꼬리 우세
    return uPct > lwPct ? 'S2' : 'S3';
}

function getPositionCode(prevLow: number, prevHigh: number, currLow: number, currHigh: number): string {
    // 뒤가 앞 안에 완전 포함
    if (prevLow <= currLow && currHigh <= prevHigh) return 'P0';
    // 뒤가 앞을 완전히 덮음
    if (currLow < prevLow && prevHigh < currHigh) return 'P1';
    // 위쪽 걸침
    if (prevLow <= currLow && currLow < prevHigh && currHigh > prevHigh) return 'P2';
    // 위로 분리
    if (currLow >= prevHigh) return 'P3';
    // 아래쪽 걸침
    if (currLow < prevLow && prevLow < currHigh && currHigh <= prevHigh) return 'P4';
    // 아래로 분리
    if (currHigh <= prevLow) return 'P5';

    return 'P0';  // 기본값
}

function getQCode(prevVol: number, currVol: number): string {
    if (prevVol === 0) return 'Q3';

    const volRel = (currVol / prevVol) * 100;

    if (volRel < 40) return 'Q0';
    if (volRel < 70) return 'Q1';
    if (volRel < 90) return 'Q2';
    if (volRel <= 110) return 'Q3';
    if (volRel <= 150) return 'Q4';
    if (volRel <= 250) return 'Q5';
    return 'Q6';
}

// 3. 메인 변환 함수
export function candleToHangul(candles: Candle[]): CandleResult[] {
    const results: CandleResult[] = [];
    const N = Math.min(60, candles.length);

    for (let i = 0; i < candles.length; i++) {
        const candle = candles[i];

        // range_ma 계산
        let rangeMa: number;
        if (i < N) {
            rangeMa = candles.slice(0, i + 1).reduce((sum, c) => sum + (c.high - c.low), 0) / (i + 1);
        } else {
            rangeMa = candles.slice(i - N + 1, i + 1).reduce((sum, c) => sum + (c.high - c.low), 0) / N;
        }

        // 캔들 자체 코드
        const V = getVCode(candle, rangeMa);
        const D = getDCode(candle);
        const S = getSCode(candle);

        // 캔들 자체 한글
        const candleChar = combineHangul(
            V_CHOSUNG[V],
            D_JUNGSUNG[D],
            S_JONGSUNG[S]
        );

        // 관계 코드
        let PB: string, Q: string, PR: string;

        if (i === 0) {
            // 첫 캔들
            PB = 'PB_NONE';
            Q = 'Q_NONE';
            PR = 'PR_NONE';
        } else {
            const prev = candles[i - 1];

            // 몸통 위치
            const prevBodyLow = Math.min(prev.open, prev.close);
            const prevBodyHigh = Math.max(prev.open, prev.close);
            const currBodyLow = Math.min(candle.open, candle.close);
            const currBodyHigh = Math.max(candle.open, candle.close);
            PB = getPositionCode(prevBodyLow, prevBodyHigh, currBodyLow, currBodyHigh).replace('P', 'PB');

            // 레인지 위치
            PR = getPositionCode(prev.low, prev.high, candle.low, candle.high).replace('P', 'PR');

            // 거래량
            Q = getQCode(prev.jdiff_vol, candle.jdiff_vol);
        }

        // 관계 한글
        const relationChar = combineHangul(
            PB_CHOSUNG[PB],
            Q_JUNGSUNG[Q],
            PR_JONGSUNG[PR]
        );

        results.push({
            date: candle.date,
            time: candle.time,
            candle_code: `${V}-${D}-${S}`,
            relation_code: `${PB}-${Q}-${PR}`,
            candle_char: candleChar,
            relation_char: relationChar,
            hangul: `${candleChar}${relationChar}`,
            raw: candle
        });
    }

    return results;
}

// 한글 시퀀스를 날짜별로 줄바꿈하여 반환하는 헬퍼 함수
export function formatHangulByDate(results: CandleResult[], debug: boolean = false): string {
    if (results.length === 0) return '';

    const lines: string[] = [];
    let currentDate = results[0].date;
    let currentLine = results[0].hangul;

    if (debug) {
        console.log(`[DEBUG] 캔들 0: 날짜=${results[0].date}, 시간=${results[0].time}, 한글=${results[0].hangul}`);
    }

    for (let i = 1; i < results.length; i++) {
        if (debug) {
            console.log(`[DEBUG] 캔들 ${i}: 날짜=${results[i].date}, 시간=${results[i].time}, 한글=${results[i].hangul}`);
        }

        if (results[i].date !== currentDate) {
            // 날짜가 바뀌면 현재 라인을 저장하고 새 라인 시작
            if (debug) {
                console.log(`[DEBUG] *** 날짜 변경 감지: ${currentDate} -> ${results[i].date} ***`);
                console.log(`[DEBUG] 현재 라인 저장: ${currentLine}`);
            }
            lines.push(currentLine);
            currentDate = results[i].date;
            currentLine = results[i].hangul;
        } else {
            // 같은 날짜면 스페이스와 함께 이어붙이기
            currentLine += ' ' + results[i].hangul;
            if (debug) {
                console.log(`[DEBUG] 같은 날짜, 이어붙임: ${currentLine}`);
            }
        }
    }

    // 마지막 라인 추가
    if (debug) {
        console.log(`[DEBUG] 마지막 라인 저장: ${currentLine}`);
    }
    lines.push(currentLine);

    if (debug) {
        console.log(`[DEBUG] 총 ${lines.length}개 라인 생성`);
    }

    return lines.join('\n');
}
